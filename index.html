// ASP.NET Core 8/9 — Clean Architecture & CQRS Controller
[ApiController]
[Route("api/v1/[controller]")]
[Authorize]
public class OrdersController : ControllerBase
{
    private readonly ISender _mediator;
    private readonly ILogger<OrdersController> _logger;

    public OrdersController(ISender mediator, ILogger<OrdersController> logger)
    {
        _mediator = mediator;
        _logger = logger;
    }

    /// <summary>Creates a new order with idempotent transaction guarantee</summary>
    [HttpPost]
    [ProducesResponseType(typeof(OrderResponseDto), 201)]
    public async Task<IActionResult> CreateOrder([FromBody] CreateOrderCommand command)
    {
        var result = await _mediator.Send(command);
        
        if (result.IsFailure)
            return BadRequest(new { error = result.Error.Message });

        return CreatedAtAction(nameof(GetOrderById), new { id = result.Value.Id }, result.Value);
    }
}
// FinTech Integration — Secure HMAC & Webhook Idempotency
public class StripeWebhookService : IPaymentWebhookHandler
{
    private readonly IUnitOfWork _unitOfWork;
    private readonly IWhatsAppService _whatsapp;

    public async Task<Result> ProcessWebhookAsync(string jsonPayload, string signatureHeader)
    {
        // 1. Verify cryptographic signature from payment provider
        var stripeEvent = EventUtility.ConstructEvent(jsonPayload, signatureHeader, _webhookSecret);

        // 2. Prevent duplicate processing via DB Idempotency Key
        if (await _unitOfWork.ProcessedEvents.ExistsAsync(stripeEvent.Id))
            return Result.Success(); // Already handled

        // 3. Process payment intent & trigger instant notifications
        if (stripeEvent.Type == Events.PaymentIntentSucceeded)
        {
            var intent = (PaymentIntent)stripeEvent.Data.Object;
            await _unitOfWork.Orders.MarkPaidAsync(intent.Metadata["OrderId"]);
            await _whatsapp.SendOrderConfirmationAsync(intent.CustomerEmail);
        }

        await _unitOfWork.SaveChangesAsync();
        return Result.Success();
    }
}
// Angular 17+ — Reactive State with Signals & Clean Services
@Component({
  selector: 'app-analytics-dashboard',
  standalone: true,
  imports: [CommonModule, CurrencyPipe],
  template: `
    <div class="metrics-grid">
      <div class="metric-card">
        <span>Active Revenue</span>
        <h3>{{ totalRevenue() | currency:'USD' }}</h3>
      </div>
    </div>
  `
})
export class AnalyticsDashboardComponent implements OnInit {
  private readonly metricService = inject(MetricService);

  // Fine-grained reactive signals
  readonly metrics = signal<ProjectMetric[]>([]);
  readonly totalRevenue = computed(() => 
    this.metrics().reduce((sum, m) => sum + m.amount, 0)
  );

  ngOnInit(): void {
    this.metricService.getLiveMetrics()
      .subscribe(data => this.metrics.set(data));
  }
}
