// OrdersController.cs — ASP.NET Core Web API
[Authorize]
[ApiController]
[Route("api/[controller]")]
public class OrdersController : ControllerBase
{
    private readonly IOrderService _orderService;

    public OrdersController(IOrderService orderService)
    {
        _orderService = orderService;
    }

    // TODO: add pagination parameters before production deployment
    [HttpGet]
    public async Task<IActionResult> GetRecentOrders([FromQuery] int limit = 20)
    {
        var orders = await _orderService.GetRecentAsync(limit);
        return Ok(orders);
    }
}
// analytics.service.ts — Angular 17+ / RxJS
@Injectable({ providedIn: 'root' })
export class AnalyticsService {
  private readonly http = inject(HttpClient);
  private readonly apiUrl = '/api/analytics';

  // TODO: add retry logic for intermittent connection drops
  getDashboardSummary(): Observable<DashboardSummary> {
    return this.http.get<DashboardSummary>(`${this.apiUrl}/summary`).pipe(
      catchError(err => {
        console.error('Failed to fetch dashboard metrics', err);
        return throwError(() => err);
      })
    );
  }
}
